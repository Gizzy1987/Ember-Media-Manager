CREATE VIEW episodelist
        AS
        SELECT
        episode.*,
        (SELECT files.strFilename FROM files WHERE files.idFile = episode.idFile) AS strFilePath,
        (SELECT source.strName FROM tvshowsource AS source WHERE source.idSource = episode.idSource) AS Source,
        (SELECT fanart.url FROM art AS fanart WHERE fanart.media_id = episode.idEpisode AND fanart.media_type = 'episode' AND fanart.type = 'fanart' LIMIT 1) AS FanartPath,
        (SELECT poster.url FROM art AS poster WHERE poster.media_id = episode.idEpisode AND poster.media_type = 'episode' AND poster.type = 'thumb' LIMIT 1) AS PosterPath,
        (SELECT GROUP_CONCAT(DISTINCT credits.strActor) FROM writerlinkepisode AS writerlink LEFT OUTER JOIN actors AS credits ON credits.idActor = writerlink.idWriter WHERE writerlink.idEpisode = episode.idEpisode) AS Credits,
        (SELECT GROUP_CONCAT(DISTINCT directors.strActor) FROM directorlinkepisode AS directorlink LEFT OUTER JOIN actors AS directors ON directors.idActor = directorlink.idDirector WHERE directorlink.idEpisode = episode.idEpisode) AS Director,
        CASE WHEN episode.idFile IS -1 THEN 1 ELSE 0 END AS Missing
        FROM
        episode;

CREATE VIEW movielist
        AS
        SELECT
        movie.*,
        CASE WHEN movie.SortTitle IS NOT '' THEN movie.SortTitle ELSE movie.Title END AS SortedTitle,
        (SELECT source.strName FROM moviesource AS source WHERE source.idSource = movie.idSource) AS Source,
        (SELECT banner.url FROM art AS banner WHERE banner.media_id = movie.idMovie AND banner.media_type = 'movie' AND banner.type = 'banner' LIMIT 1) AS BannerPath,
        (SELECT clearart.url FROM art AS clearart WHERE clearart.media_id = movie.idMovie AND clearart.media_type = 'movie' AND clearart.type = 'clearart' LIMIT 1) AS ClearArtPath,
        (SELECT clearlogo.url FROM art AS clearlogo WHERE clearlogo.media_id = movie.idMovie AND clearlogo.media_type = 'movie' AND clearlogo.type = 'clearlogo' LIMIT 1) AS ClearLogoPath,
        (SELECT discart.url FROM art AS discart WHERE discart.media_id = movie.idMovie AND discart.media_type = 'movie' AND discart.type = 'discart' LIMIT 1) AS DiscArtPath,
        (SELECT fanart.url FROM art AS fanart WHERE fanart.media_id = movie.idMovie AND fanart.media_type = 'movie' AND fanart.type = 'fanart' LIMIT 1) AS FanartPath,
        (SELECT keyart.url FROM art AS keyart WHERE keyart.media_id = movie.idMovie AND keyart.media_type = 'movie' AND keyart.type = 'keyart' LIMIT 1) AS KeyartPath,
        (SELECT landscape.url FROM art AS landscape WHERE landscape.media_id = movie.idMovie AND landscape.media_type = 'movie' AND landscape.type = 'landscape' LIMIT 1) AS LandscapePath,
        (SELECT poster.url FROM art AS poster WHERE poster.media_id = movie.idMovie AND poster.media_type = 'movie' AND poster.type = 'poster' LIMIT 1) AS PosterPath,
        (SELECT GROUP_CONCAT(DISTINCT countries.strCountry) FROM countrylinkmovie AS countrylink LEFT OUTER JOIN country AS countries ON countries.idCountry = countrylink.idCountry WHERE countrylink.idMovie = movie.idMovie) AS Country,
        (SELECT GROUP_CONCAT(DISTINCT credits.strActor) FROM writerlinkmovie AS writerlink LEFT OUTER JOIN actors AS credits ON credits.idActor = writerlink.idWriter WHERE writerlink.idMovie = movie.idMovie) AS Credits,
        (SELECT GROUP_CONCAT(DISTINCT directors.strActor) FROM directorlinkmovie AS directorlink LEFT OUTER JOIN actors AS directors ON directors.idActor = directorlink.idDirector WHERE directorlink.idMovie = movie.idMovie) AS Director,
        (SELECT GROUP_CONCAT(DISTINCT genres.strGenre) FROM genrelinkmovie AS genrelink LEFT OUTER JOIN genre AS genres ON genres.idGenre = genrelink.idGenre WHERE genrelink.idMovie = movie.idMovie) AS Genre,
        (SELECT GROUP_CONCAT(DISTINCT studios.strStudio) FROM studiolinkmovie AS studiolink LEFT OUTER JOIN studio AS studios ON studios.idStudio = studiolink.idStudio WHERE studiolink.idMovie = movie.idMovie) AS Studio,
        (SELECT GROUP_CONCAT(DISTINCT tags.strTag) FROM taglinks AS taglink LEFT OUTER JOIN tag AS tags ON tags.idTag = taglink.idTag WHERE taglink.idMedia = movie.idMovie AND taglink.media_type = 'movie') AS 'Tag'
        FROM
        movie;

CREATE VIEW setslist

        SELECT
        sets.*,
        (SELECT banner.url FROM art AS banner WHERE banner.media_id = sets.idSet AND banner.media_type = 'set' AND banner.type = 'banner' LIMIT 1) AS BannerPath,
        (SELECT clearart.url FROM art AS clearart WHERE clearart.media_id = sets.idSet AND clearart.media_type = 'set' AND clearart.type = 'clearart' LIMIT 1) AS ClearArtPath,
        (SELECT clearlogo.url FROM art AS clearlogo WHERE clearlogo.media_id = sets.idSet AND clearlogo.media_type = 'set' AND clearlogo.type = 'clearlogo' LIMIT 1) AS ClearLogoPath,
        (SELECT discart.url FROM art AS discart WHERE discart.media_id = sets.idSet AND discart.media_type = 'set' AND discart.type = 'discart' LIMIT 1) AS DiscArtPath,
        (SELECT fanart.url FROM art AS fanart WHERE fanart.media_id = sets.idSet AND fanart.media_type = 'set' AND fanart.type = 'fanart' LIMIT 1) AS FanartPath,
        (SELECT keyart.url FROM art AS keyart WHERE keyart.media_id = sets.idSet AND keyart.media_type = 'set' AND keyart.type = 'keyart' LIMIT 1) AS KeyartPath,
        (SELECT landscape.url FROM art AS landscape WHERE landscape.media_id = sets.idSet AND landscape.media_type = 'set' AND landscape.type = 'landscape' LIMIT 1) AS LandscapePath,
        (SELECT poster.url FROM art AS poster WHERE poster.media_id = sets.idSet AND poster.media_type = 'set' AND poster.type = 'poster' LIMIT 1) AS PosterPath,
        (SELECT COUNT(setlink.idMovie) FROM setlinkmovie AS setlink WHERE setlink.idSet = sets.idSet) AS 'Count',
        (SELECT GROUP_CONCAT(DISTINCT movies.Title) FROM setlinkmovie AS setlink LEFT OUTER JOIN movie AS movies ON movies.idMovie = setlink.idMovie WHERE setlink.idSet = sets.idSet) AS 'MovieTitles'
        FROM
        sets;

CREATE VIEW seasonslist
        AS
        SELECT
        seasons.*,
        (SELECT banner.url FROM art AS banner WHERE banner.media_id = seasons.idSeason AND banner.media_type = 'season' AND banner.type = 'banner' LIMIT 1) AS BannerPath,
        (SELECT fanart.url FROM art AS fanart WHERE fanart.media_id = seasons.idSeason AND fanart.media_type = 'season' AND fanart.type = 'fanart' LIMIT 1) AS FanartPath,
        (SELECT landscape.url FROM art AS landscape WHERE landscape.media_id = seasons.idSeason AND landscape.media_type = 'season' AND landscape.type = 'landscape' LIMIT 1) AS LandscapePath,
        (SELECT poster.url FROM art AS poster WHERE poster.media_id = seasons.idSeason AND poster.media_type = 'season' AND poster.type = 'poster' LIMIT 1) AS PosterPath,
        CASE WHEN seasons.Season IS NOT -1 THEN (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1) ELSE NULL END AS Episodes,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1 AND e.Playcount IS NOT NULL) AS Playcount,
        CASE WHEN seasons.Season IS NOT -1 THEN CASE WHEN (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1) IS NOT 0 AND (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1) = (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1 AND e.Playcount IS NOT NULL) THEN 1 ELSE 0 END ELSE NULL END AS HasWatched,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1 AND e."New" IS 1) AS NewEpisodes,
        CASE WHEN (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = seasons.idShow AND e.Season = seasons.Season AND e.idFile IS NOT -1) IS NOT 0 THEN 0 ELSE 1 END AS Missing
        FROM
        seasons;

CREATE VIEW tvshowlist
        AS
        SELECT
        tvshow.*,
        CASE WHEN tvshow.SortTitle IS NOT '' THEN tvshow.SortTitle ELSE tvshow.Title END AS SortedTitle,
        (SELECT source.strName FROM tvshowsource AS source WHERE source.idSource = tvshow.idSource) AS Source,
        (SELECT banner.url FROM art AS banner WHERE banner.media_id = tvshow.idShow AND banner.media_type = 'tvshow' AND banner.type = 'banner' LIMIT 1) AS BannerPath,
        (SELECT characterart.url FROM art AS characterart WHERE characterart.media_id = tvshow.idShow AND characterart.media_type = 'tvshow' AND characterart.type = 'characterart' LIMIT 1) AS CharacterArtPath,
        (SELECT clearart.url FROM art AS clearart WHERE clearart.media_id = tvshow.idShow AND clearart.media_type = 'tvshow' AND clearart.type = 'clearart' LIMIT 1) AS ClearArtPath,
        (SELECT clearlogo.url FROM art AS clearlogo WHERE clearlogo.media_id = tvshow.idShow AND clearlogo.media_type = 'tvshow' AND clearlogo.type = 'clearlogo' LIMIT 1) AS ClearLogoPath,
        (SELECT fanart.url FROM art AS fanart WHERE fanart.media_id = tvshow.idShow AND fanart.media_type = 'tvshow' AND fanart.type = 'fanart' LIMIT 1) AS FanartPath,
        (SELECT keyart.url FROM art AS keyart WHERE keyart.media_id = tvshow.idShow AND keyart.media_type = 'tvshow' AND keyart.type = 'keyart' LIMIT 1) AS KeyartPath,
        (SELECT landscape.url FROM art AS landscape WHERE landscape.media_id = tvshow.idShow AND landscape.media_type = 'tvshow' AND landscape.type = 'landscape' LIMIT 1) AS LandscapePath,
        (SELECT poster.url FROM art AS poster WHERE poster.media_id = tvshow.idShow AND poster.media_type = 'tvshow' AND poster.type = 'poster' LIMIT 1) AS PosterPath,
        (SELECT GROUP_CONCAT(DISTINCT countries.strCountry) FROM countrylinktvshow AS countrylink LEFT OUTER JOIN country AS countries ON countries.idCountry = countrylink.idCountry WHERE countrylink.idShow = tvshow.idShow) AS Country,
        (SELECT GROUP_CONCAT(DISTINCT creators.strActor) FROM creatorlinktvshow AS creatorlink LEFT OUTER JOIN actors AS creators ON creators.idActor = creatorlink.idActor WHERE creatorlink.idShow = tvshow.idShow) AS Creator,
        (SELECT GROUP_CONCAT(DISTINCT directors.strActor) FROM directorlinktvshow AS directorlink LEFT OUTER JOIN actors AS directors ON directors.idActor = directorlink.idDirector WHERE directorlink.idShow = tvshow.idShow) AS Director,
        (SELECT GROUP_CONCAT(DISTINCT genres.strGenre) FROM genrelinktvshow AS genrelink LEFT OUTER JOIN genre AS genres ON genres.idGenre = genrelink.idGenre WHERE genrelink.idShow = tvshow.idShow) AS Genre,
        (SELECT GROUP_CONCAT(DISTINCT studios.strStudio) FROM studiolinktvshow AS studiolink LEFT OUTER JOIN studio AS studios ON studios.idStudio = studiolink.idStudio WHERE studiolink.idShow = tvshow.idShow) AS Studio,
        (SELECT GROUP_CONCAT(DISTINCT tags.strTag) FROM taglinks AS taglink LEFT OUTER JOIN tag AS tags ON tags.idTag = taglink.idTag WHERE taglink.idMedia = tvshow.idShow AND taglink.media_type = 'tvshow') AS Tag,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1) AS Episodes,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1 AND e.Playcount IS NOT NULL) AS Playcount,
        CASE WHEN (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1) IS NOT 0 AND (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1) = (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1 AND e.Playcount IS NOT NULL) THEN 1 ELSE 0 END AS HasWatched,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1 AND e."New" IS 1) AS NewEpisodes,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1 AND e.Mark IS 1) AS MarkedEpisodes,
        (SELECT COUNT(DISTINCT e.idEpisode) FROM episode AS e WHERE e.idShow = tvshow.idShow AND e.idFile IS NOT -1 AND e.Lock IS 1) AS LockedEpisodes
        FROM
        tvshow;
